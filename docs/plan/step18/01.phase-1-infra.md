**파일**: `src/utils/config-loader.js`

**변경 내용**:

기존 auth 기본값 정규화 블록 (`config.auth = { ... }`) 뒤에 추가:

```javascript
// signupMode 정규화
if (!['approval', 'self'].includes(config.auth.signupMode)) {
  if (config.auth.signupMode) {
    console.warn(`Warning: auth.signupMode "${config.auth.signupMode}" is invalid. Using default "approval".`);
  }
  config.auth.signupMode = 'approval';
}

// selfSignup 기본값
config.auth.selfSignup = config.auth.selfSignup || {};
config.auth.selfSignup.defaultGroupName = config.auth.selfSignup.defaultGroupName || 'Viewer';
```

**검증 포인트**:
- [x] `signupMode` 미설정 → `"approval"`
- [x] `signupMode: "invalid"` → 경고 로그 + `"approval"` 폴백
- [x] `selfSignup` 미설정 → `{ defaultGroupName: "Viewer" }`

### 1.4 activity-logger.js 신규 생성

**파일**: `src/utils/activity-logger.js` (신규)

**전체 구현**:

```javascript
'use strict';

/**
 * Activity Logger — 카테고리별 활동 로그 기록
 * 기존 logger 인스턴스를 래핑하여 일관된 포맷 제공
 *
 * 로그 포맷: [CATEGORY] EVENT key=value key=value ...
 */

let logger = null;

function init(loggerInstance) {
  logger = loggerInstance;
}

function formatDetails(details) {
  return Object.entries(details)
    .filter(([, v]) => v !== undefined && v !== null)
    .map(([k, v]) => {
      if (typeof v === 'object') return `${k}=${JSON.stringify(v)}`;
      return `${k}=${v}`;
    })
    .join(' ');
}

function log(category, event, details) {
  if (!logger) return;
  const msg = `[${category}] ${event} ${formatDetails(details)}`;
  logger.info(msg);
}

function logWarn(category, event, details) {
  if (!logger) return;
  const msg = `[${category}] ${event} ${formatDetails(details)}`;
  logger.warn(msg);
}

function logError(category, event, details) {
  if (!logger) return;
  const msg = `[${category}] ${event} ${formatDetails(details)}`;
  logger.error(msg);
}

// 마스킹 유틸: API 키 앞 8자만 표시
function maskKey(key) {
  if (!key || key.length <= 8) return key;
  return key.substring(0, 8) + '...';
}

// 요청에서 사용자 정보 추출
function extractUser(req) {
  if (req.session?.email) return req.session.email;
  if (req.adminSession?.email) return req.adminSession.email;
  if (req.adminSession?.name) return req.adminSession.name;
  if (req.authUser?.email) return req.authUser?.email;
  if (req.authUser?.name) return `apikey(${maskKey(req.authUser.name)})`;
  return 'anonymous';
}

function extractIp(req) {
  return req.ip || req.connection?.remoteAddress || 'unknown';
}

module.exports = {
  init,
  maskKey,
  extractUser,
  extractIp,
  auth:  (event, details) => log('AUTH', event, details),
  doc:   (event, details) => log('DOC', event, details),
  mcp:   (event, details) => log('MCP', event, details),
  admin: (event, details) => log('ADMIN', event, details),
  authWarn:  (event, details) => logWarn('AUTH', event, details),
  authError: (event, details) => logError('AUTH', event, details),
  mcpError:  (event, details) => logError('MCP', event, details),
};
```

## 테스트 시나리오

### 정상 케이스

**TC-1-01**: config에 signupMode 설정 시 정상 로딩

- **Given**: `config.json5`에 `auth.signupMode: "self"` 설정
- **When**: 서버 시작
- **Then**: `config.auth.signupMode === "self"`, `config.auth.selfSignup.defaultGroupName === "Viewer"`

**TC-1-02**: activity-logger 로그 기록

- **Given**: 서버 시작 후 activity-logger 초기화됨
- **When**: `activityLogger.auth('LOGIN', { email: 'test@co.com', ip: '1.2.3.4' })` 호출
- **Then**: 로그 파일에 `[AUTH] LOGIN email=test@co.com ip=1.2.3.4` 기록

### 예외 케이스

**TC-1-03**: signupMode 유효하지 않은 값

- **Given**: `config.json5`에 `auth.signupMode: "invalid"`
- **When**: 서버 시작
- **Then**: 경고 로그 출력, `config.auth.signupMode === "approval"` 폴백

**TC-1-04**: selfSignup 미설정

- **Given**: `config.json5`에 `auth.selfSignup` 없음
- **When**: 서버 시작
- **Then**: `config.auth.selfSignup.defaultGroupName === "Viewer"`

### 경계값

**TC-1-05**: activity-logger 초기화 전 호출

- **Given**: `activityLogger.init()` 호출 전
- **When**: `activityLogger.auth(...)` 호출
- **Then**: 에러 없이 무시됨 (logger가 null)
